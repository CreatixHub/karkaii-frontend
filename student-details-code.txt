import { useCallback, useState } from 'react';
import { Link } from 'react-router-dom';
import ToastStack from '@components/ui/ToastStack';
import { all_routes } from '@/routes/all_routes';
import { useReveal } from '@hooks/useReveal';
import { useToasts } from '@hooks/useToasts';
import { downloadTextFile } from '@utils/download.utils';
import ExecutiveOverview from './components/ExecutiveOverview';
import ProfileHero from './components/ProfileHero';
import ProfileSidebar from './components/ProfileSidebar';
import ProfileTabs from './components/ProfileTabs';
import type { ProfilePaneId } from './components/ProfileTabs';
import OverviewPane from './components/panes/OverviewPane';
import AttendancePane from './components/panes/AttendancePane';
import AcademicsPane from './components/panes/AcademicsPane';
import ExamsPane from './components/panes/ExamsPane';
import FeesPane from './components/panes/FeesPane';
import type { PaymentTarget } from './components/panes/FeesPane';
import TimetablePane from './components/panes/TimetablePane';
import HealthPane from './components/panes/HealthPane';
import LibraryPane from './components/panes/LibraryPane';
import DocumentsPane from './components/panes/DocumentsPane';
import TimelinePane from './components/panes/TimelinePane';
import AchievementsPane from './components/panes/AchievementsPane';
import DisciplinePane from './components/panes/DisciplinePane';

import NotesPane from './components/panes/NotesPane';

import ProfileSkeleton from './components/ProfileSkeleton';
import type { ConfirmOptions, ProfileAction, ProfileModalId } from './data/studentProfileActions';
import {
  ACTION_TOASTS,
  CONFIRMATIONS,
  WHATSAPP_URL,
} from './data/studentProfileActions';
import {
  CERTIFICATE_TEXT,
  PROFILE_TEXT,
  REPORT_CARD_TEXT,
  recordLinks,
} from './data/studentProfile.data';

const routes = all_routes;

/** Receipt body generated after a successful payment, as the script built it. */
const receiptText = (label: string, amount: number, txn: string) =>
  `KARKAII ACADEMY - PAYMENT RECEIPT
====================================
Receipt No : ${txn}
Student    : Janet Daniel (AD1256589) - Class III-A
Paid For   : ${label}
Amount     : $${amount.toLocaleString()}
Method     : Card ****4521
Status     : PAID - Confirmed
Payer      : Robert Daniel (Parent)
====================================
System-generated receipt.
`;

const StudentDetails = () => {
  const [pane, setPane] = useState<ProfilePaneId>('overview');
  const [activeModal, setActiveModal] = useState<ProfileModalId | null>(null);
  const [confirm, setConfirm] = useState<ConfirmOptions | null>(null);
  const [payment, setPayment] = useState<PaymentTarget>({
    label: 'Term II Tuition',
    amount: 2000,
  });
  const [moreOpen, setMoreOpen] = useState(false);

  const { loading, revealed } = useReveal();
  const { toasts, push, dismiss } = useToasts();

  const openModal = useCallback((modal: ProfileModalId) => setActiveModal(modal), []);
  const closeModal = useCallback(() => setActiveModal(null), []);

  /** `download_doc` in the script: slugify the row's name and write a stub file. */
  const downloadDoc = useCallback(
    (name: string) => {
      downloadTextFile(
        `${name.replace(/[^a-z0-9]+/gi, '-')}.txt`,
        `KARKAII ACADEMY\n${name}\nStudent: Janet Daniel (AD1256589)\n`,
      );
      push(`Downloading ${name}…`, 'success');
    },
    [push],
  );

  const copyText = useCallback(
    (value: string) => {
      if (!value) return;
      navigator.clipboard?.writeText(value).then(
        () => push('Copied to clipboard.', 'success'),
        () => push('Copied to clipboard.', 'success'),
      );
    },
    [push],
  );

  /** Mirrors the `data-sp-action` switch in student-profile.js. */
  const handleAction = useCallback(
    (action: ProfileAction) => {
      const toast = ACTION_TOASTS[action];
      if (toast) push(toast.message, toast.variant);

      switch (action) {
        case 'print':
          window.print();
          return;
        case 'message':
        case 'email':
        case 'call':
        case 'meeting':
        case 'homework':
        case 'idcard':
          openModal(action);
          return;
        case 'track_bus':
          openModal('track');
          return;
        case 'whatsapp':
          window.open(WHATSAPP_URL, '_blank', 'noopener');
          return;
        case 'download':
          downloadTextFile('Janet-Daniel-AD1256589.txt', PROFILE_TEXT);
          return;
        case 'certificate':
          downloadTextFile('Certificate-Janet-Daniel.txt', CERTIFICATE_TEXT);
          return;
        case 'download_report':
          downloadTextFile('Report-Card-Janet-Daniel-TermII.txt', REPORT_CARD_TEXT);
          return;
        case 'transfer':
        case 'suspend':
        case 'archive':
        case 'reset_password':
          setConfirm(CONFIRMATIONS[action]);
          openModal('confirm');
          return;
        default:
          // `upload` only toasts, which already happened above.
          return;
      }
    },
    [openModal, push],
  );

  const confirmAction = useCallback(() => {
    if (confirm) push(confirm.toast.message, confirm.toast.variant);
    setConfirm(null);
    closeModal();
  }, [confirm, closeModal, push]);

  const startPayment = useCallback(
    (target: PaymentTarget) => {
      setPayment(target);
      openModal('pay');
    },
    [openModal],
  );

  /** The script generated a receipt file 1.35 s after a successful payment. */
  const onPaid = useCallback((target: PaymentTarget) => {
    const txn = `TXN${target.label.length * 7919 + String(target.amount).length * 1049 + 100237}`;
    downloadTextFile(`Receipt-${txn}.txt`, receiptText(target.label, target.amount, txn));
  }, []);

  const renderPane = () => {
    switch (pane) {
      case 'overview':
        return <OverviewPane />;
      case 'attendance':
        return <AttendancePane revealed={revealed} />;
      case 'academics':
        return <AcademicsPane revealed={revealed} />;
      case 'exams':
        return (
          <ExamsPane revealed={revealed} onAction={handleAction} onDownloadDoc={downloadDoc} />
        );
      case 'fees':
        return (
          <FeesPane
            revealed={revealed}
            onPay={startPayment}
            onOpenModal={openModal}
            onDownloadDoc={downloadDoc}
          />
        );
      case 'timetable':
        return <TimetablePane />;
      case 'health':
        return <HealthPane onAction={handleAction} onDownloadDoc={downloadDoc} />;
      case 'library':
        return <LibraryPane revealed={revealed} onRenew={downloadDoc} />;
      case 'documents':
        return <DocumentsPane onAction={handleAction} onDownloadDoc={downloadDoc} />;
      case 'timeline':
        return <TimelinePane />;
      case 'achievements':
        return <AchievementsPane revealed={revealed} />;
      case 'discipline':
        return <DisciplinePane revealed={revealed} />;
      case 'communication':
        return <CommunicationPane revealed={revealed} onAction={handleAction} />;
      case 'notes':
        return <NotesPane onToast={push} />;
      case 'audit':
        return <AuditPane revealed={revealed} />;
      default:
        return null;
    }
  };

  return (
    <>
      {/* Page Wrapper */}
      <div className="page-wrapper">
        <div className="content">
          {/* Page Header */}
          <div className="d-md-flex d-block align-items-center justify-content-between mb-3 flex-wrap gap-2">
            <div className="my-auto mb-2 d-flex align-items-center gap-2">
              <Link to={routes.studentGrid} className="profile-back" aria-label="Back to students">
                <i className="ti ti-arrow-left" />
              </Link>
              <div>
                <h3 className="page-title mb-1">Student Profile</h3>
                <nav>
                  <ol className="breadcrumb mb-0">
                    <li className="breadcrumb-item">
                      <Link to={routes.adminDashboard}>Dashboard</Link>
                    </li>
                    <li className="breadcrumb-item">
                      <Link to={routes.studentGrid}>Student</Link>
                    </li>
                    <li className="breadcrumb-item active" aria-current="page">
                      Student 360° Profile
                    </li>
                  </ol>
                </nav>
              </div>
            </div>

            <div className="student-profile">
              <div className="profile-actions my-auto">
                <Link to={routes.editStudent} className="profile-button profile-button-primary">
                  <i className="ti ti-edit" />
                  Edit Student
                </Link>
                <button
                  type="button"
                  className="profile-button"
                  onClick={() => handleAction('print')}
                >
                  <i className="ti ti-printer" />
                  Print
                </button>
                <button
                  type="button"
                  className="profile-button"
                  onClick={() => handleAction('download')}
                >
                  <i className="ti ti-download" />
                  Download
                </button>
                <button
                  type="button"
                  className="profile-button"
                  onClick={() => handleAction('idcard')}
                >
                  <i className="ti ti-id" />
                  ID Card
                </button>

                <div className="dropdown">
                  <button
                    type="button"
                    className="profile-button profile-icon-button"
                    aria-label="More actions"
                    aria-expanded={moreOpen}
                    onBlur={() => window.setTimeout(() => setMoreOpen(false), 150)}
                    onClick={() => setMoreOpen((open) => !open)}
                  >
                    <i className="ti ti-dots-vertical" />
                  </button>
                  <ul className={`dropdown-menu dropdown-menu-end p-2${moreOpen ? ' show' : ''}`}>
                    <li>
                      <h4 className="dropdown-header dropdown-head fs-15 fw-medium">
                        Student Records
                      </h4>
                    </li>
                    {recordLinks.map((link) => (
                      <li key={link.id}>
                        <Link
                          className="dropdown-item rounded-1"
                          to={routes[link.route as keyof typeof routes]}
                          onClick={() => setMoreOpen(false)}
                        >
                          <i className={`${link.icon} me-2`} />
                          {link.label}
                        </Link>
                      </li>
                    ))}
                    <li>
                      <hr className="dropdown-divider" />
                    </li>
                    <li>
                      <h5 className="dropdown-header dropdown-head fs-15">Actions</h5>
                    </li>
                    <li>
                      <Link
                        className="dropdown-item rounded-1"
                        to={routes.studentPromotion}
                        onClick={() => setMoreOpen(false)}
                      >
                        <i className="ti ti-arrow-ramp-right-2 me-2" />
                        Promote Student
                      </Link>
                    </li>
                    <li>
                      <button
                        type="button"
                        className="dropdown-item rounded-1"
                        onClick={() => {
                          setMoreOpen(false);
                          handleAction('transfer');
                        }}
                      >
                        <i className="ti ti-transfer me-2" />
                        Transfer Student
                      </button>
                    </li>
                    <li>
                      <button
                        type="button"
                        className="dropdown-item rounded-1"
                        onClick={() => {
                          setMoreOpen(false);
                          openModal('login');
                        }}
                      >
                        <i className="ti ti-lock me-2" />
                        Login Details
                      </button>
                    </li>
                    <li>
                      <hr className="dropdown-divider" />
                    </li>
                    <li>
                      <button
                        type="button"
                        className="dropdown-item rounded-1"
                        onClick={() => {
                          setMoreOpen(false);
                          handleAction('suspend');
                        }}
                      >
                        <i className="ti ti-ban me-2" />
                        Suspend Student
                      </button>
                    </li>
                    <li>
                      <button
                        type="button"
                        className="dropdown-item rounded-1"
                        onClick={() => {
                          setMoreOpen(false);
                          handleAction('archive');
                        }}
                      >
                        <i className="ti ti-archive me-2" />
                        Archive Student
                      </button>
                    </li>
                  </ul>
                </div>
              </div>
            </div>
          </div>
          {/* /Page Header */}

          <div className={`student-profile${loading ? '' : ' loaded'}`} id="studentProfile">
            {loading ? <ProfileSkeleton /> : null}

            {!loading ? (
              <div className="profile-live">
                <ProfileHero revealed={revealed} onAction={handleAction} />
                <ExecutiveOverview revealed={revealed} />

                <div className="row g-3">
                  <ProfileSidebar onAction={handleAction} />

                  <div className="col-xxl-8 col-xl-7">
                    <div className="card sticky-item">
                      <div className="card-header">
                        <ProfileTabs active={pane} onChange={setPane} />
                      </div>
                      <div className="card-body">{renderPane()}</div>
                    </div>
                  </div>
                </div>
              </div>
            ) : null}
          </div>
        </div>
      </div>
      {/* /Page Wrapper */}

      <StudentProfileModals
        active={activeModal}
        onClose={closeModal}
        onToast={push}
        onAction={handleAction}
        confirm={confirm}
        onConfirm={confirmAction}
        payment={payment}
        onPaid={onPaid}
      />

      <TrackBusModal
        open={activeModal === 'track'}
        onClose={closeModal}
        onAction={handleAction}
      />

      <LoginDetailsModal
        open={activeModal === 'login'}
        onClose={closeModal}
        onCopy={copyText}
        onAction={handleAction}
      />

      <ToastStack toasts={toasts} onDismiss={dismiss} id="sp_toasts" />
    </>
  );
};

export default StudentDetails;

import { useEffect, useRef } from 'react';
import { Link } from 'react-router-dom';
import ImageWithBasePath from '@core/common/imageWithBasePath';
import { all_routes } from '@/routes/all_routes';

import type { StudentModalId } from '../data/studentsWorkspace.data';

const routes = all_routes;

export interface StudentCardItemProps {
  student: StudentCard;
  selected: boolean;
  onToggleSelect: (id: string) => void;
  menuOpen: boolean;
  onToggleMenu: (id: string | null) => void;
  onOpenModal: (modal: StudentModalId) => void;
  onPreview: (student: StudentCard) => void;
}

const StudentCardItem = ({
  student,
  selected,
  onToggleSelect,
  menuOpen,
  onToggleMenu,
  onOpenModal,
  onPreview,
}: StudentCardItemProps) => {
  const menuRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    if (!menuOpen) return;
    const close = (event: MouseEvent) => {
      if (!menuRef.current?.contains(event.target as Node)) onToggleMenu(null);
    };
    document.addEventListener('mousedown', close);
    return () => document.removeEventListener('mousedown', close);
  }, [menuOpen, onToggleMenu]);

  const closeThen = (run: () => void) => () => {
    onToggleMenu(null);
    run();
  };

  return (
    <div className="col-xxl-3 col-xl-4 col-md-6 d-flex">
      <div className="card flex-fill mb-0">
        <div className="card-header d-flex align-items-center justify-content-between gap-2">
          <div className="d-flex align-items-center gap-2 overflow-hidden">
            <div className="form-check mb-0">
              <input
                className="form-check-input"
                type="checkbox"
                id={`pick_${student.admissionNo}`}
                aria-label={`Select ${student.name}`}
                checked={selected}
                onChange={() => onToggleSelect(student.id)}
              />
            </div>
            <span className="fs-12 fw-medium text-muted text-truncate">
              {student.admissionNo}
            </span>
          </div>

            <span className={`d-inline-flex align-items-center fs-12 fw-medium text-${student.status.color}`}>

              {student.status.label}
            </span>

                type="button"
                className="btn btn-icon btn-sm p-0 border-0 bg-transparent text-muted"
                aria-expanded={menuOpen}
                aria-label="Student actions"
                onClick={() => onToggleMenu(menuOpen ? null : student.id)}
              >
                <i className="ti ti-dots-vertical" />
              </button>

              <ul className={`dropdown-menu dropdown-menu-end p-2${menuOpen ? ' show' : ''}`}>
                <li className="context-menu-head">{student.name}</li>

/* Types are taken from the data arrays themselves, so they always match the data file. */
type Item<T> = T extends readonly (infer U)[] ? U : never;
type DetailField = Item<typeof personalDetails>;
type GuardianCard = Item<typeof guardians>;
type TimelineEntry = Item<typeof admissionTimeline>;
type ActivityEntry = Item<typeof activityHistory>;

interface ApiHistory {
  id: string | number;
  fromStatus: string | null;
  toStatus: string;
  note: string | null;
  changedByName: string | null;
  createdAt: string;
}

/** GET /api/admissions/:id */
export interface ApiAdmissionDetail {

  applicationNo: string;

  lastName: string;

  phone?: string | null;
  bloodGroup?: string | null;
  fatherName?: string | null;

  fatherEmail?: string | null;
  fatherOccupation?: string | null;
  motherName?: string | null;
  motherPhone?: string | null;
  motherEmail?: string | null;
  motherOccupation?: string | null;

  guardianPhone: string;
  guardianEmail?: string | null;
  previousSchool?: string | null;
  previousClass?: string | null;
  addressLine1?: string | null;
  addressLine2?: string | null;
  postalCode?: string | null;
  status: string;
  appliedOn: string;
  decisionNotes?: string | null;
  needsTransport?: boolean;
  needsHostel?: boolean;

  feePaid?: boolean;
  source: string;

  history?: ApiHistory[];

const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];

const dash = (v: unknown): string => {
  const s = v === null || v === undefined ? '' : String(v).trim();
  return s ? s : '-';
};

const pretty = (iso: string): string => {
  const m = /^(\d{4})-(\d{2})-(\d{2})/.exec(iso ?? '');
  return m ? `${m[3]} ${MONTHS[Number(m[2]) - 1] ?? m[2]} ${m[1]}` : dash(iso);
};

const when = (iso: string): string => {
  const d = new Date(iso);
  if (Number.isNaN(d.getTime())) return dash(iso);
  return d.toLocaleString('en-GB', { day: '2-digit', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit' });
};

const humanize = (s: string): string => {
  const t = s.replace(/_/g, ' ');
  return t.charAt(0).toUpperCase() + t.slice(1);
};

const SOURCE_LABEL: Record<string, string> = {
  walk_in: 'Walk-in',
  website: 'Website',
  referral: 'Referral',
  agent: 'Agent',
  event: 'Event',
  other: 'Other',
};

function statusBadge(api: string): { label: string; className: string } {
  switch (api) {
    case 'accepted':
      return { label: 'Approved', className: 'badge-soft-success' };
    case 'rejected':
      return { label: 'Rejected', className: 'badge-soft-danger' };
    case 'withdrawn':
      return { label: 'Withdrawn', className: 'badge-soft-danger' };
    case 'waitlisted':
      return { label: 'Waitlisted', className: 'badge-soft-warning' };
    case 'document_review':
    case 'shortlisted':
    case 'offered':
      return { label: 'Under Review', className: 'badge-soft-primary' };
    default:
      return { label: 'Pending', className: 'badge-soft-warning' };
  }
}

/** The first password of a new student login: DD-MM-YYYY of the date of birth. */
export function dobPasswordText(dob: string): string {
  const [y, m, d] = (dob ?? '').split('-');
  return y && m && d ? `${d}-${m}-${y}` : '';
}

function replace<T>(target: T[], items: T[]) {
  target.splice(0, target.length, ...items);
}

export function loadAdmissionDetail(id: string): Promise<ApiAdmissionDetail> {
  return apiClient.get<ApiAdmissionDetail>(endpoints.admissions.detail(id));
}

/** Overwrites the page's data objects (in place) with one real application. */
export function applyAdmission(a: ApiAdmissionDetail): void {
  const st = statusBadge(a.status);
  const fullName = [a.firstName, a.lastName].filter(Boolean).join(' ');
  const dob = pretty(a.dob);
  const applied = pretty(a.appliedOn);
  const counselor = a.assignedToStaffName ?? 'Unassigned';
  const feeLabel = a.feePaid ? 'Paid' : 'Unpaid';

  Object.assign(profile, {
    name: fullName,
    admissionNo: a.applicationNo,
    status: st,
    classApplied: dash(a.className),
    academicYear: '-',
    appliedDate: applied,
    parentName: a.guardianName,
    parentPhone: a.guardianPhone,
  });

  for (const card of summaryCards as unknown as Array<Record<string, unknown>>) {
    switch (card.id) {
      case 'status':
        card.value = st.label;
        card.valueClassName = st.className;
        break;
      case 'stage':
        card.value = humanize(a.status);
        break;
      case 'fee':
        card.value = feeLabel;
        card.valueClassName = a.feePaid ? 'badge-soft-success' : 'badge-soft-danger';
        break;
      case 'interview':
        card.value = 'Not Scheduled';
        break;
      case 'documents':
        card.value = 'None uploaded';
        break;
      case 'counselor':
        card.value = counselor;
        break;
      default:
        break;
    }
  }

  const filled = [a.dob, a.gender, a.phone, a.bloodGroup, a.addressLine1, a.previousSchool, a.guardianPhone, a.className, a.email];
  Object.assign(overview, {
    completeness: Math.round((filled.filter((v) => v && String(v).trim()).length / filled.length) * 100),
    basics: [
      { label: 'Class Applied', value: dash(a.className) },
      { label: 'Academic Year', value: '-' },
      { label: 'Gender', value: dash(a.gender) },
      { label: 'Date of Birth', value: dob },
      { label: 'Applied On', value: applied },
      { label: 'Assigned Staff', value: counselor },
    ],
  });

  replace<DetailField>(personalDetails, [
    { label: 'First Name', value: dash(a.firstName) },
    { label: 'Last Name', value: dash(a.lastName) },
    { label: 'Gender', value: dash(a.gender) },
    { label: 'Date of Birth', value: dob },
    { label: 'Blood Group', value: dash(a.bloodGroup) },
    { label: 'Student Phone (login username)', value: dash(a.phone) },
    { label: 'Email', value: dash(a.email) },
  ] as DetailField[]);
  setPersonalAddress([a.addressLine1, a.addressLine2, a.postalCode].filter((v) => v && String(v).trim()).join(', ') || '-');

  replace<DetailField>(academicDetails, [
    { label: 'Class Applied', value: dash(a.className) },
    { label: 'Application No', value: a.applicationNo },
    { label: 'Source', value: SOURCE_LABEL[a.source] ?? dash(a.source) },
    { label: 'Academic Year', value: '-' },
  ] as DetailField[]);

  replace<DetailField>(medicalDetails, [
    { label: 'Blood Group', value: dash(a.bloodGroup) },
    { label: 'Allergies', value: '-' },
    { label: 'Medical Conditions', value: '-' },
    { label: 'Medications', value: '-' },
  ] as DetailField[]);
  setMedicalDoctorContact('-');

  const cards: GuardianCard[] = [
    {
      name: a.guardianName,
      role: 'Guardian',
      avatar: 'assets/img/profiles/avatar-27.jpg',
      phone: a.guardianPhone,
      email: dash(a.guardianEmail),
      occupation: '-',
    },
  ] as GuardianCard[];
  if (a.fatherName) {
    cards.push({
      name: a.fatherName,
      role: 'Father',
      avatar: 'assets/img/profiles/avatar-27.jpg',
      phone: dash(a.fatherPhone),
      email: dash(a.fatherEmail),
      occupation: dash(a.fatherOccupation),
    } as GuardianCard);
  }
  if (a.motherName) {
    cards.push({
      name: a.motherName,
      role: 'Mother',
      avatar: 'assets/img/profiles/avatar-23.jpg',
      phone: dash(a.motherPhone),
      email: dash(a.motherEmail),
      occupation: dash(a.motherOccupation),
    } as GuardianCard);
  }
  replace<GuardianCard>(guardians, cards);

  replace<DetailField>(emergencyDetails, [
    { label: 'Contact Name', value: a.guardianName },
    { label: 'Relationship', value: 'Guardian' },
    { label: 'Phone', value: a.guardianPhone },
    { label: 'Alternate Phone', value: '-' },
  ] as DetailField[]);

  replace<DetailField>(transportDetails, [
    { label: 'Transport Required', value: a.needsTransport ? 'Yes' : 'No' },
    { label: 'Route', value: '-' },
    { label: 'Pickup Point', value: '-' },
    { label: 'Vehicle', value: '-' },
  ] as DetailField[]);

  replace<DetailField>(hostelDetails, [
    { label: 'Hostel Required', value: a.needsHostel ? 'Yes' : 'No' },
    { label: 'Hostel Name', value: '-' },
    { label: 'Room Type', value: '-' },
  ] as DetailField[]);

  replace<DetailField>(previousSchoolDetails, [
    { label: 'School Name', value: dash(a.previousSchool) },
    { label: 'Board', value: '-' },
    { label: 'Last Class', value: dash(a.previousClass) },
    { label: 'TC Number', value: '-' },
  ] as DetailField[]);

  documentCards.splice(0, documentCards.length);

  const fee = a.applicationFee !== null && a.applicationFee !== undefined ? Number(a.applicationFee) : null;
  Object.assign(feeSummary, {
    total: fee === null ? '-' : fee.toFixed(2),
    paid: fee === null ? '-' : a.feePaid ? fee.toFixed(2) : '0.00',
    balance: fee === null ? '-' : a.feePaid ? '0.00' : fee.toFixed(2),
  });

  const hist = a.history ?? [];
  const timeline = hist.map((h) => ({
    title: h.fromStatus ? `Status: ${humanize(h.toStatus)}` : 'Application Submitted',
    meta: `by ${h.changedByName ?? 'System'} - ${when(h.createdAt)}`,
    icon: 'ti ti-check',
    tone: 'success',
  })) as unknown as TimelineEntry[];
  if (a.status !== 'accepted') {
    timeline.push({ title: 'Enrollment', meta: 'Pending', icon: 'ti ti-user-check', tone: 'pending' } as unknown as TimelineEntry);
  }
  replace<TimelineEntry>(admissionTimeline, timeline);

  notes.splice(
    0,
    notes.length,
    ...((a.decisionNotes
      ? [{ avatar: 'assets/img/profiles/avatar-27.jpg', text: a.decisionNotes, meta: 'Decision note' }]
      : []) as unknown as typeof notes),
  );

  replace<ActivityEntry>(
    activityHistory,
    [...hist].reverse().map((h) => ({
      icon: 'ti ti-activity',
      iconAccent: 'bg-soft-primary',
      title: h.note ? `${humanize(h.toStatus)}: ${h.note}` : `Status changed to ${humanize(h.toStatus)}`,
      meta: `${h.changedByName ?? 'System'} - ${when(h.createdAt)}`,
    })) as unknown as ActivityEntry[],
  );
}
